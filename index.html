// Query 1 â€” CovidDeaths_Cleaned + 
// Load, type, filter, and enrich the source data
let
  Source = Excel.Workbook(
      File.Contents("D:\New folder\DATA PROJECT 2024\New folder (2)\CovidDeaths.xlsx"),
      null, true),

  Sheet       = Source{[Item="Sheet1",Kind="Sheet"]}[Data],
  Headers     = Table.PromoteHeaders(Sheet, [PromoteAllScalars=true]),

  TypedData   = Table.TransformColumnTypes(Headers, {
      {"iso_code", type text},   {"continent", type text},
      {"location", type text},   {"date", type date},
      {"total_cases", Int64.Type}, {"new_cases", Int64.Type},
      {"total_deaths", Int64.Type},{"new_deaths", Int64.Type},
      {"total_deaths_per_million", type number},
      {"population", Int64.Type}}),

  // Remove continent-level aggregate rows (no iso continent code)
  FilteredRows  = Table.SelectRows(TypedData, each
      [continent] <> null and [continent] <> ""),

  YearMonth   = Table.AddColumn(FilteredRows, "YearMonth",
      each Date.ToText([date], "yyyy-MM"), type text),
  Year        = Table.AddColumn(YearMonth, "Year",
      each Date.Year([date]), Int64.Type),
  MonthNum    = Table.AddColumn(Year, "MonthNum",
      each Date.Month([date]), Int64.Type),
  CFR         = Table.AddColumn(MonthNum, "CFR_Percent",
      each if [total_cases] <> null and [total_cases] > 0
           then Number.Round(([total_deaths] / [total_cases]) * 100, 2)
           else null, type number)
in
  CFR

      each Number.Round([TotalDeaths] / 3180223 * 100, 1),
      type number),

      {{"TotalDeaths", Order.Descending}})

  Sorted
// Query 3 â€” Top10_Countries +
